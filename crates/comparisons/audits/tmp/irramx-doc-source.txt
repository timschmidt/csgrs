75 // The original file is buggy in the case when the function is a constant
function. (it never stops!)

76 // this should be reported sometime.

[ 77](namespacei_r_r_a_m.html#a84a945e008ba7ede5c2235e27058777d) int
[module](namespacei_r_r_a_m.html#a84a945e008ba7ede5c2235e27058777d)(std::function<REAL(REAL)>
f, const REAL &x, int p)

78 {

79 // Semantics: If m=module(f,x,p), then

80 // |x-z| <=2^m implies |f(x)-f(z)| <= 2^p

81

82 // If we are able to approximate f(x) by a DYADIC number d with an error of
<=2^{p-1},

83 // then for any z in the argument interval of x, f(z) must differ by at
most

84 // 2^{p-1} from d, hence |f(x)-f(z)|<=2^p

85

86 int result;

87 if (get_cached(result))

88 return result;

89

90 DYADIC d;

91 REAL x_copy = x;

92

93 sizetype argerror, testerror;

94

95 x_copy.geterror(argerror);

96 sizetype_set(testerror, 1, argerror.exponent);

97 x_copy.adderror(testerror);

98 {

99 single_valued code;

100 d = approx(f(x_copy), p - 1);

101 }

102 // At this line, we are sure that x_copy (and so also x) is precise enough
to allow

103 // the computation of f(x), even with a slightly increased error of the
argument.

104

105 // We now try to find the smallest p_arg such that the evaluation of f(x+-
2^p_arg)

106 // is possible up to an error of at most 2^{p-1}

107

108 // ITERATION_STACK SAVED_STACK;

109

110 // To do this, we start with p_arg=p.

111 // If this is successfull, we increase the value of p_arg until the first
failure

112 // It it is not, then we decrease until the first success...

113 int direction = 0, p_arg = p;

114 bool try_it = true;

115

116 while (try_it)

117 {

118

119 sizetype_set(testerror, 1, p_arg);

120 x_copy.seterror(argerror);

121 x_copy.adderror(testerror);

122 bool fail = false;

123 if (iRRAM_unlikely(state->debug > 0))

124 {

125 sizetype x_error;

126 x_copy.geterror(x_error);

127 iRRAM_DEBUG2(1, "Testing module: 1*2^%d + %d*2^%d\n", p_arg,
argerror.mantissa, argerror.exponent);

128 iRRAM_DEBUG2(1, "argument error: %d*2^%d\n", x_error.mantissa,
x_error.exponent);

129 }

130 try

131 {

132 single_valued code;

133 REAL z = f(x_copy);

134 if (iRRAM_unlikely(state->debug > 0))

135 {

136 sizetype z_error;

137 z.geterror(z_error);

138 iRRAM_DEBUG2(1, "Module yields result %d*2^%d\n", z_error.mantissa,
z_error.exponent);

139 }

140 d = approx(z, p - 1);

141 }

142 catch (Iteration it)

143 {

144 fail = true;

145 }

146 switch (direction)

147 {

148 case 0:;

149 if (fail)

150 direction = -1;

151 else

152 direction = 1;

153 p_arg += direction;

154 break;

155 case 1:;

156 if (fail || p_arg > 0)

157 {

158 try_it = false;

159 p_arg -= direction;

160 }

161 else

162 {

163 p_arg += direction;

164 };

165 break;

166 case -1:;

167 if (fail)

168 {

169 p_arg += direction;

170 }

171 else

172 {

173 try_it = false;

174 };

175 break;

176 }

177 }

178 iRRAM_DEBUG2(1, "Modules resulting in p_arg=%d\n", p_arg);

179

180 sizetype_set(testerror, 1, p_arg);

181 sizetype_inc(argerror, testerror);

182

183 while (argerror.mantissa > 1)

184 {

185 argerror.mantissa = argerror.mantissa >> 1;

186 argerror.exponent += 1;

187 }

188

189 result = argerror.exponent;

190 put_cached(result);

191 return result;

192 }

[module2_](namespacei_r_r_a_m.html#aeadf1ea271178a5303daf4dc98568434)(std::function<[Point<N>](namespacei_r_r_a_m.html#a1d2e14e811b03f40a0e28611c957bd64)([Point<M>](namespacei_r_r_a_m.html#a1d2e14e811b03f40a0e28611c957bd64))>

204 sizetype err;

205

206 // create H

207 [HyperCube<M>](namespacei_r_r_a_m.html#a47a131eda86521c2be784e9c7f712a7a)
box = c;

208 for(REAL &u : box) {

209 sizetype_set(err, 1, -s-1);

210 u.seterror(err);

211 }

212

213 // check if f(H') is subset of a hypercube

214 DYADIC d;

215 bool success = true;

216 try {

217 single_valued code;

218

219 [HyperCube<N>](namespacei_r_r_a_m.html#a47a131eda86521c2be784e9c7f712a7a)
fBox = f(box);

220 for(REAL &v : fBox) d = approx(v, -p-1);

221 } catch (Iteration it) { success = false; }

222

223 // return current one if success

224 if(success) return s;

225

226 // bisection on failure, total 2^M sub-hypercubes

227 int result = s+1;

228 [HyperCube<M>](namespacei_r_r_a_m.html#a47a131eda86521c2be784e9c7f712a7a)
offset, newCenter;

229 REAL halfLen =
[Exp](namespacei_r_r_a_m.html#a5811649b09be686cc0a6db277e2a6d85)(-s-1);

230 offset.fill(0);

231 for(int k=0 ; k<(1<<M) ; k++) {

232 // configure offset

233 // If the j'th bit of k is 0, we choose the left half section on j'th
dimension

234 // Otherwise, the right half section.

235 for(int j=0 ; j<M ; j++) {

236 int sign = ((k & (1<<j))<<1)-1; // -1 on left section, 1 otherwise

237 offset[j] = REAL(INTEGER(sign)) * halfLen;

238 }

239

240 // configure point

241 for(int j=0 ; j<M ; j++) newCenter[j] = c[j] + offset[j];

242

243 // find the required precision

244 result = max(result, module2_<M,N>(f, p, newCenter, s+1));

245 }

246

247 return result;

248 }

47 // Find the pArg

48 // must |f(u)-f(z)| < (2^-p)/sqrt(2) to include the box with a ball

49 // hence find find pArg such that whenever |x-z| < 2^-pArg, |f(x)-f(z)| <
