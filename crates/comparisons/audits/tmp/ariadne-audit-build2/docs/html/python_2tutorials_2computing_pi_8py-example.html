28def compute_pi():
29 # There are many formulae for pi.
30 # We will use the identity tan(pi/6)=1/sqrt(3),
31 # yielding pi=6*atan(1/sqrt(3))
32
33 r = 6*atan(rec(sqrt(3))) # rec(x) is the reciprocal 1/x
34
35 print("r:",r,type(r)) # r has type Real, and is represented by a symbolic formula
36
37 # Compute r to an accuracy of 20 binary digits of precision (approximately 6 decimal places)
38 acc = Accuracy(1/two**20)
39 print("acc:",acc)
40 v=r.compute(acc)
41 print("v:",v,type(v))
42
43 # The result is stored as a Dyadic lower and upper bound for r.
44 # To output a double-precision approversion
45 pr=double_precision
46 pr=DoublePrecision() # Alternative syntax using constructor
47 pr=dp # Alternative syntax using abbreviation
48 x=v.get(pr)
49 print("x:",x,type(x))
50
51 print("sin(x/6):",sin(x/6),type(sin(x/6)))
52 print("sin(x/4):",sin(x/4))
53
54 print()
55
56 # Compute r to an accuracy of 1/2^120 binary digits of precision (approximately 40 decimal places)
57 acc=Accuracy(Dyadic(1,120))
58 print("acc:",acc)
59 v=r.compute(acc)
60 print("v:",v,type(v))
61
62 # To output double-precision approximation
63 pr=double_precision
64 x=v.get(pr)
65 print("x:",x,type(x))
66
67 # To output multiple-precision approversion with 128bits
68 pr=precision(128)
69 pr=multiple_precision(128)
70 pr=MultiplePrecision(128)
71 pr=MP(128)
72 x=v.get(pr)
73 print("x:",x,type(x))
74 print("repr(x):",repr(x),type(x))
75 print("x.precision():",x.precision())
76 print("sin(x/6):",sin(x/6))
77 print("sin(x/4):",sin(x/4))
78
79 print()
