28def function_demonstration():
29 #! [Function demonstration]
30
31 # Create a user-defined scalar-valued function
32 x=[EffectiveScalarMultivariateFunction.coordinate(2,index) for index in range(0,2)]
33 sf=sqrt(sqr(x[0])+sqr(x[1]))
34 print("sf:",sf)
35 a=FloatDPApproximationVector([4,3],dp)
36 sfa=sf(a)
37 print("sf(a):",sfa)
38 b=FloatDPBoundsVector([4,3],dp)
39 sfb=sf(b)
40 print("sf(b):",sfb)
41
42 x=EffectiveVectorMultivariateFunction.identity(3)
43 vf=EffectiveVectorMultivariateFunction([sqrt(sqr(x[0])+sqr(x[1])),x[1]+x[2]])
44 print("vf:",vf)
45 a=FloatDPApproximationVector([4,3,0],dp)
46 vfa=vf(a)
47 print("vf(a):",vfa)
48 b=FloatDPBoundsVector([4,3,0],dp)
49 vfb=evaluate(vf,b)
50 print("evaluate(vf,b):",vfb)
51
52 dsf0=derivative(sf,0);
53 print("derivative(sf,0):",dsf0);
54
55 p0=FloatDPApproximationMultivariatePolynomial.coordinate(2,0,dp);
56 p1=FloatDPApproximationMultivariatePolynomial.coordinate(2,1,dp);
57 print("p1:",p1)
58 a3=FloatDPApproximation(3,dp)
59 a5=FloatDPApproximation(5,dp)
60 q=p0*(p0+p1*a3)+a5
61 print("q:",q)

65def calculus_demonstration():
66 #! [Calculus demonstration]
67
68 # Create a box to act as the domain of a Taylor function
69 dom=BoxDomainType([{4:7},{1:6},{-1:+1}])
70 print("dom:",dom)
71
72 # Create a sweeper to control the accuracy of a Taylor function
73 swp=ThresholdSweeperDP(dp,1e-8)
74 #swp=GradedSweeper(dp,6)
75 print("swp:",swp)
76
77 # Create the scalar Taylor model representing a constant function with given value on the domain dom
78 tc=ValidatedScalarMultivariateTaylorFunctionModelDP.constant(dom,dec_(4.2),swp)
79 print("tc:",tc,"\n")
80
81 # Create the scalar Taylor model representing the function x1 on the domain dom
82 t1=ValidatedScalarMultivariateTaylorFunctionModelDP.coordinate(dom,1,swp)
83 print("t1:",t1,"\n")
84
85 # Create all Taylor variables on the domain dom
86 t=[ValidatedScalarMultivariateTaylorFunctionModelDP.coordinate(dom,i,swp) for i in range(0,3)]
87 print("t[0]:",t[0])
88 print("t[1]:",t[1])
89 print("t[2]:",t[2])
90
91 # Make shorthands for the variable names
92 tx=t[0]
93 ty=t[1]
94 tz=t[2]
95 print("tx:",tx)
96 print("ty:",ty)
97 print("tz:",tz)
98
99 #Create a ValidatedScalarMultivariateTaylorFunctionModelDP from a Function
100 p=EffectiveScalarMultivariateFunction.coordinate(3,0)
101 tp=ValidatedScalarMultivariateTaylorFunctionModelDP(dom,p,swp)
102 print("p:",p)
103 print("tp:",tp)
104
105 # The domain D of tx.
106 print("tx.domain():",tx.domain())
107 # A not-necessarily tight over-approximation to p(D)+/-e.
108 print("tx.codomain():",tx.codomain())
109 # An over-approximation to p(D)+/-e.
110 print("tx.range():",tx.range())
111
112 # Define some constants
113 n=int(3)
114 c=FloatDP(exact(0.75),dp)
115 b=FloatDPBounds(exact(0.625),exact(0.875),dp);
116
117 # Arithmetic on Taylor models
118 +tx; -tx; tx+ty; tx-ty; tx*ty; tx/ty;
119
120 # Mixed arithmetic
121 tx+c; tx-c; tx*c; tx/c;
122 c+tx; c-tx; c*tx; c/tx;
123
124 tx+b; tx-b; tx*b; tx/b;
125 b+tx; b-tx; b*tx; b/tx;
126
127 # Lattice functions
128 min(tx,ty); max(tx,ty); abs(tx);
129
130 # Univariate arithmetical functions
131 neg(tx);
132 rec(tx);
133 sqr(tx);
134 pow(tx,n);
135
136 # Algebraic and transcendental functions
137 sqrt(tx);
138 exp(tx); log(tx);
139 sin(tx), cos(tx), tan(tx/2)
140
141 # Inplace operations
142 tx+=ty; tx-=ty;
143 tx+=c; tx-=c; tx*=c; tx/=c;
144 tx+=b; tx-=b; tx*=b; tx/=b;
145
146 # Set coordinate functions tx0, tx1
147 dom=BoxDomainType([{-1:+1},{-1:+1}])
148 tx0=ValidatedScalarMultivariateTaylorFunctionModelDP.coordinate(dom,0,swp);
149 tx1=ValidatedScalarMultivariateTaylorFunctionModelDP.coordinate(dom,1,swp);
150
151 tf=ValidatedScalarMultivariateTaylorFunctionModelDP(1+2*tx0+3*tx0*tx1+tx0*tx0)
152 # Compute an antiderivative of f with respect to x[0] whose value is zero at the midpoint of the domain of x[0]
153 Itf0=antiderivative(tf,0)
154 # Compute the antiderivative of f with respect to x[1] whose value is zero when x[1]=c
155 Itf1=antiderivative(tf,1,c)
156 print("antiderivative(tf,0):",Itf0)
157 print("antiderivative(tf,1):",Itf1)
158
159 # Restrict to a subdomain
160 subdom=BoxDomainType([{exact(-0.125):exact(0.125)},{exact(0.5):exact(0.75)}])
161 w=tx0*tx1
162 rw=restriction(w,subdom)
163 print("restriction(w,subdom):",rw)
164
165 #Join tf and tg into a VectorTaylorFunction x:->[tf(x),tg(x)]
166 tg=ValidatedVectorMultivariateTaylorFunctionModelDP([exp(tx0)*cos(tx1)])
167 tfg=join(tf,tg)
168 print("join(tf,tg):",tfg)
169
170 # Combine tf and tg into a VectorTaylorFunction (x,y):->[tf(x),tg(y)]
171 tfg=combine(tf,tg)
172 print("combine(tf,tg):",tfg)
173
174 # Function composition
175 dom=BoxDomainType([{4:7},{1:6},{-1:1}])
176 th=ValidatedVectorMultivariateTaylorFunctionModelDP.identity(dom,swp)
177 cd=th.codomain()
178
179 f=EffectiveScalarMultivariateFunction.coordinate(3,0)
180 g=EffectiveVectorMultivariateFunction.zeros(2,3)
181
182 tf=ValidatedScalarMultivariateTaylorFunctionModelDP(cd,f,swp)
183 tg=ValidatedVectorMultivariateTaylorFunctionModelDP(cd,g,swp)
184
185 # Compose an scalar multivariate function and a vector Taylor function
186 fth=compose(f,th)
187 print("compose(f,th):",fth)
188
189 # Compose a vector multivariate function and a vector Taylor function
190 gth=compose(g,th)
191 print("compose(g,th):",gth)
192
193 # Compose a scalar and a vector Taylor function
194 assert(subset(th.codomain(),tf.domain()))
195 tfh=compose(tf,th)
196 print("compose(tf,th):",tfh)
197
198 # Compose two vector Taylor functions
199 assert(subset(th.codomain(),tg.domain()))
200 tgh=compose(tg,th)
201 print("compose(tg,th):",tgh)
202
203
204 # Consistency and refinement checking
205 dom=BoxDomainType([{-1:+1},{-1:+1}]);
206 tf1=tx0+tx0*tx0/2
207 tf2=tx0+FloatDPBounds(-1,1,dp)
208 print("tf1:",tf1)
209 print("tf2:",tf1)
210 ic=inconsistent(tf1,tf2)
211 print("inconsistent(tf1,tf2):",ic)
212 rf=refines(tf1,tf2)
213 print("refines(tf1,tf2):",rf)
214 tf=refinement(tf1,tf2)
215 print("refinement(tf1,tf2):",tf)
216 #! [Calculus demonstration]
217
