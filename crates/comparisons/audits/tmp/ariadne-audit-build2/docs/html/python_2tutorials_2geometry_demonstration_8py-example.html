31 # Create intervals with different endpoint types
32 ivlq = RationalInterval(Rational(4,7),Rational(3,5))
33 ivlq = Interval[Rational](Rational(4,7),Rational(3,5))
34 print("ivlq:",ivlq,type(ivlq))
35 ivlu = FloatDPUpperInterval({exact(1.5):exact(2.5)})
36 ivlu = Interval[FloatDPUpperBound]({exact(1.5):exact(2.5)})
37 print("ivlu:",ivlu,type(ivlu))
38
39 # Create a box from a list of intervals
40 # Interval literals can be given as [l,u] or (for int/float values) as {l:u}
41 bx = RealBox([ [Rational(4,7),Rational(3,5)] , {exact(1.5):exact(2.5)} ])
42 bx = Box[Real]([ [Rational(4,7),Rational(3,5)] , {exact(1.5):exact(2.5)} ])
43 print("bx",bx,type(bx))
44
45 # Test geometric predicates
46 subset({2:3},{1:4})
47 disjoint({2:3},{4:5})
48
49 # Create sets based on constraints
50 x = EffectiveScalarMultivariateFunction.coordinate(2,0)
51 y = EffectiveScalarMultivariateFunction.coordinate(2,1)
52 cs = ConstraintSet([sqr(x)+2*sqr(y)<=1,3*x+2*y>=1])
53 print("cs:",cs)
54
55 bx = Box[Real]([[-1,+1],[-1,+1]])
56 # Compute the intersection of a contraint set with a box, obtaining a bounded set
57 bcs = intersection(bx,cs)
58 print("bcs:",bcs,type(bcs))
59
60 # Compute the image of a bounded contraint set under a continuous function
61 h=EffectiveVectorMultivariateFunction([dec_(1.5)-x*x-y/3,y])
62 cis = image(bcs,h)
63 print("cis:",cis,type(cis))
64
65 # Discretise the constrained image set on a grid
66 g=Grid(2)
67 dpth=4
68 gtp = outer_approximation(cis,g,dpth)
69 print("gtp:",gtp,type(gtp))
70
71
72 (cis1,cis2)=cis.split()
73 (cis11,cis12)=cis1.split()
74 (cis21,cis22)=cis2.split()
75
76 # Plot an approximation to the set
77 prj=Projection2d(2,0,1) # The identity projection
78 wnd=[{0:3},{-2:+2}] # The view window
79 green=Colour(0,1,0)
80 plot("geometry_demonstration", prj, wnd, [(green,cis)])
81 #! [Geometry demonstration]
