34 # Number classes and their constructors
35
36 # Create an integer
37 z=Integer(5)
38 print("Integer(5):",z)
39
40 # Create a dyadic; can convert from Integer
41 w=Dyadic(z)
42 w=Dyadic(5)
43 w=Dyadic(11,3) # Constructs the number 11/2^3=11/8
44 w=11/two**3 # Alternative syntax for 11/2^3
45 print("Dyadic(11,3):",w)
46
47 # Create a decimal number; can convert from Dyadic
48 g=Decimal(w)
49 g=Decimal(9.81)
50 g=Decimal("9.81")
51 print("Decimal(9.81):",g,type(g))
52
53 # Create a rational; can convert from Dyadic, Decimal
54 q=Rational(w);
55 q=Rational(5);
56 q=Rational(11,8);
57 print("Rational(11,8):",q,type(q))
58
59 # Create a real number
60 r=Real(q)
61 print("Real(4*atan(1)):",r)
62 print()
63
64 # Operations on real numbers
65 # Arithmetic operators
66 +r; -r; r+r; r-r; r*r; r/r
67 # Aliases for arithmetic operators (currently not fully supported)
68 pos(r); neg(r); # add(r,r); sub(r,r); mul(r,r); div(r,r);
69 # Other arithmetic operations
70 nul(r); sqr(r); hlf(r); rec(r); pow(r,-3)
71 # Algebraic and transcendental operations
72 sqrt(r); exp(r); log(r); sin(r); cos(r); tan(r); atan(r)
73 # Lattice operations
74 abs(r); max(r,r); min(r,r)
75 # Comparison operators
76 k=(r<=r)

81 # Store a double-precision floating-point number
82 d=ExactDouble(1.375)

93 # Define shorthands for defining Ariadne values from input

101 dp=double_precision

104 # Create a raw double-precision number
105 xdp=FloatDP(exact(1.75),dp)

112 # Create double-precision bounds for a value
113 xdpb=FloatDPBounds(Decimal(1.2),dp) # Creates the interval [1.19999999999999996:1.20000000000000018]
