26def get_tank():
27 # Declare the system constants
28 alpha = RealConstant("alpha",dec_(0.02))
29 beta = RealConstant("beta",dec_(0.3))
30
31 # Declare the variables for the dynamics
32 aperture = RealVariable("aperture")
33 height = RealVariable("height")
34
35 # Create the tank automaton
36 automaton = HybridAutomaton()
37
38 # The water level is always given by the same dynamic.
39 # The inflow is controlled by the valve aperture, the outflow depends on the
40 # pressure, which is proportional to the water height.
41 automaton.new_mode({dot(height):beta*aperture-alpha*height})

48#! [get_valve]
49def get_valve():
50 # Declare some constants. Note that system parameters should be given as variables.
51 T = RealConstant("T",4)
52
53 # Declare the shared system variables
54 aperture = RealVariable("aperture")
55
56 # Declare the events we use
57 stop_opening = DiscreteEvent("stop_opening")
58 stop_closing = DiscreteEvent("stop_closing")
59 can_open = DiscreteEvent("can_open")
60 can_close = DiscreteEvent("can_close")
61
62 # Declare the variable for the automaton name
63 valve = StringVariable("valve")
64
65 # Create the valve automaton
66 automaton = HybridAutomaton(valve.name())
67
68 # Declare the values the valve variable can have
69 opening = DiscreteLocation({valve:"opening"})
70 closed = DiscreteLocation({valve:"closed"})
71 opened = DiscreteLocation({valve:"opened"})
72 closing = DiscreteLocation({valve:"closing"})
73
74 # Define the algebraic equations for the opened/closed locations.
75 automaton.new_mode(opened,{let(aperture):1})
76 automaton.new_mode(closed,{let(aperture):0})
77 # Define the differential equations for the opening/closing locations.
78 automaton.new_mode(opening,{dot(aperture):+1/T})
79 automaton.new_mode(closing,{dot(aperture):-1/T})
80
81 # Define the transitions: source location, event and target location
82 # then a mix of reset, guard and event kind can be present; if the event kind
83 # is not specified, then also the guard can't be specified: this implicitly
84 # means that the event is an input event for this automaton.
85 automaton.new_transition(closed,can_open,opening,{next(aperture):aperture})
86 automaton.new_transition(opening,stop_opening,opened,aperture>=1,URGENT)
87 automaton.new_transition(opened,can_close,closing,{next(aperture):aperture})
88 automaton.new_transition(closing,stop_closing,closed,aperture<=0,URGENT)
89 automaton.new_transition(opening,can_close,closing,{next(aperture):aperture})
90 automaton.new_transition(closing,can_open,opening,{next(aperture):aperture})
91
92 return automaton

97#! [get_controller]
98def get_controller():
99 # Declare some constants
100 hmin = RealConstant("hmin",dec_(5.75))
101 hmax = RealConstant("hmax",dec_(7.75))
102 delta = RealConstant("delta",dec_(0.02))
103
104 # Declare the shared system variables
105 height = RealVariable("height")
106
107 # Declare the events we use
108 can_open = DiscreteEvent("can_open")
109 can_close = DiscreteEvent("can_close")
110 must_open = DiscreteEvent("must_open")
111 must_close = DiscreteEvent("must_close")
112
113 # Declare the variable for the automaton name
114 controller = StringVariable("controller")
115
116 # Create the controller automaton
117 automaton = HybridAutomaton(controller.name())
118
119 # Declare the locations for the controller
120 rising = DiscreteLocation({controller:"rising"})
121 falling = DiscreteLocation({controller:"falling"})
122
123 # Instantiate modes for each location with no dynamics
124 automaton.new_mode(rising)
125 automaton.new_mode(falling)
126
127 # Specify the invariants valid in each mode. Note that every invariant
128 # must have an action label. This is used internally, for example, to
129 # check non-blockingness of urgent actions.
130 automaton.new_invariant(falling,height>=hmin-delta,must_open)
131 automaton.new_invariant(rising,height<=hmax+delta,must_close)
132
133 # Specify the transitions, starting from the source location, according to an event, to a target location
134 # Following those arguments you specify a guard and whether the event is permissive or urgent.
135 automaton.new_transition(falling,can_open,rising,height<=hmin+delta,PERMISSIVE)
136 automaton.new_transition(rising,can_close,falling,height>=hmax-delta,PERMISSIVE)
137
138 return automaton
