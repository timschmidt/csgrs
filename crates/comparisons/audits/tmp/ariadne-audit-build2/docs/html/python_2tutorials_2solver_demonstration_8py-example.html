28def algebraic_solver_demonstration():
29 #! [Algebraic Solver demonstration]
30
31 # Compute the solution h to the vector equation f(x,h(x))=0
32 slv=IntervalNewtonSolver(1e-8,6)
33 dom=BoxDomainType([{-1:+1},{-1:+1},{-1:+1}])
34 x=EffectiveScalarMultivariateFunction.coordinate(3,0)
35 y0=EffectiveScalarMultivariateFunction.coordinate(3,1)
36 y1=EffectiveScalarMultivariateFunction.coordinate(3,2)
37 f=join(x+4*y0+y1,y0+y1)
38 print("f:",f)
39 hf=slv.implicit(f,BoxDomainType([{-1:+1}]),BoxDomainType([{-1:+1},{-1:+1}]))
40 print("implicit(f):",hf)
41
42 # Compute the solution h to the scalar equation g(x,h(x))=0
43 # with f(x,y)=4+x-y^2, so y=sqrt(4+x)
44 dom=BoxDomainType([{-1:+1},{-1:+1}])
45 x=EffectiveScalarMultivariateFunction.coordinate(2,0)
46 y=EffectiveScalarMultivariateFunction.coordinate(2,1)
47 g=x-4*y+y*y
48 print("g:",g)
49 hg=slv.implicit(g, BoxDomainType([{-1:+1}]),IntervalDomainType({-1:+1}))
50 print("implicit(g):",hg)
